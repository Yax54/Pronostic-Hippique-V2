const int kPhaseG5 = 60;
const int kTotalG5 = 3 * kPhaseG5; // 180 cycles

/// Profils de la phase A : les deux UTILES, corrélés, discrimination MODÉRÉE.
Map<String, Profil> profilsA() {
  final p = profilsNeutres(sigma: 6.0);
  p['distSpec'] = const Profil(beta: 20, gamma: 8, sigma: 6);
  p['jockey'] = const Profil(beta: 20, gamma: 8, sigma: 6);
  return p;
}

/// Profils de la phase B : jockey NÉFASTE, distSpec UTILE, MÊME corrélation.
Map<String, Profil> profilsB() {
  final p = profilsNeutres(sigma: 6.0);
  p['distSpec'] = const Profil(beta: 20, gamma: 8, sigma: 6);
  p['jockey'] = const Profil(beta: 20, gamma: -8, sigma: 6);
  return p;
}

Map<String, Profil> profilsDeG5(int cycle) {
  if (cycle < kPhaseG5) return profilsA();
  if (cycle < 2 * kPhaseG5) return profilsB();
  return profilsA();
}

String d9(double v) => v.toStringAsFixed(9);

class Trace {
  final List<double> pJ, pD;
  final List<double> mJ, mD;
  final List<int> gelJ, gelD;
  final List<double> r;
  final List<double> discJ, discD;

  const Trace({
    required this.pJ,
    required this.pD,
    required this.mJ,
    required this.mD,
    required this.gelJ,
    required this.gelD,
    required this.r,
    required this.discJ,
    required this.discD,
  });
}

Future<Trace> deroulerG5({required bool gel}) async {
  IaExperimentation.reinitialiser();
  IaExperimentation.gelActif = gel;

  final pJ = <double>[], pD = <double>[];
  final mJ = <double>[], mD = <double>[];
  final gelJ = <int>[], gelD = <int>[];
  final r = <double>[];
  final discJ = <double>[], discD = <double>[];

  final depart = projeterTable(kInitial);
  await semer(
    pronostics: <IaPronostic>[],
    poidsDepart: depart,
    autoriserVide: true,
    sigmaAttendue: 1.0,
  );

  // Amorçage + cycles, avec profils qui CHANGENT par phase.
  final courses = <IaPronostic>[
    for (var i = 0; i < kTotalG5; i++)
      pronostic(
        courseKey: 'G5_C${i.toString().padLeft(4, '0')}',
        date: dateCycle4(i, nbCycles: kTotalG5),
        resolu: false,
        variance: 6.0 + (i % 5) * 4.0,
        scoresCriteres: <String, ScoresCriteres>{
          for (var n = 1; n <= 5; n++) '$n': notesV4(n, i, profilsDeG5(i)),
        },
        arrivee: const <int>[1, 2, 3, 4, 5],
      ),
    for (var j = 0; j < 60; j++)
      pronostic(
        courseKey: 'G5_A${j.toString().padLeft(4, '0')}',
        date: dateAmorce4(j, kTotalG5),
        variance: 6.0 + (j % 5) * 4.0,
        scoresCriteres: <String, ScoresCriteres>{
          for (var n = 1; n <= 5; n++) '$n': notesV4(n, j + 1, profilsA()),
        },
        arrivee: const <int>[1, 2, 3, 4, 5],
      ),
  ];
  await semer(
    pronostics: courses,
    verifierNombre: false,
    poidsDepart: depart,
    sigmaAttendue: sommeCle(depart, kCles),
  );

  for (var i = 0; i < kTotalG5; i++) {
    final c = await executerCycle(
      courseKey: 'G5_C${i.toString().padLeft(4, '0')}',
      arrivee: const <int>[1, 2, 3, 4, 5],
    );
    final p = poidsEffectifs();
    pJ.add(p['jockey']!);
    pD.add(p['distSpec']!);
    mJ.add((c.momentum['jockey'] ?? 0.0).abs());
    mD.add((c.momentum['distSpec'] ?? 0.0).abs());
    gelJ.add(IaExperimentation.derniersGeles.contains('jockey') ? 1 : 0);
    gelD.add(IaExperimentation.derniersGeles.contains('distSpec') ? 1 : 0);
    r.add(correlationsMemorisees()['distSpec|jockey'] ?? 0.0);
    discJ.add(IaExperimentation.dernieresDiscriminations['jockey'] ?? 0.0);
    discD.add(IaExperimentation.dernieresDiscriminations['distSpec'] ?? 0.0);
  }
  IaExperimentation.reinitialiser();
  return Trace(
    pJ: pJ,
    pD: pD,
    mJ: mJ,
    mD: mD,
    gelJ: gelJ,
    gelD: gelD,
    r: r,
    discJ: discJ,
    discD: discD,
  );
}

int somme(List<int> v, int a, int b) => v.sublist(a, b).fold(0, (x, y) => x + y);
double moy(List<double> v, int a, int b) {
  final s = v.sublist(a, b);
  return s.reduce((x, y) => x + y) / s.length;
}

double maxDe(List<double> v) => v.reduce((a, b) => a > b ? a : b);
double minDe(List<double> v) => v.reduce((a, b) => a < b ? a : b);

    var cyclesConcordants = 0;
    var cyclesGelesB = 0;
    for (var i = kPhaseG5; i < 2 * kPhaseG5; i++) {
      if (avecGel.gelJ[i] == 1 || avecGel.gelD[i] == 1) {
        cyclesGelesB++;
        if (avecGel.gelJ[i] == 1 &&
            avecGel.gelD[i] == 0 &&
            avecGel.discJ[i] < 0) {
          cyclesConcordants++;
        }
      }
    }
    w('[G4v5] phase B : cycles avec au moins un gel = $cyclesGelesB');
    w('[G4v5]            cycles où c est BIEN jockey (néfaste) qui est gelé, '
        'et lui seul = $cyclesConcordants');
    if (cyclesGelesB > 0 && cyclesGelesB == cyclesConcordants) {
      w('[G4v5]            ⇒ COÏNCIDENCE TOTALE ($cyclesConcordants/$cyclesGelesB) '
          ': le critère gelé se confond, dans cette cellule, avec le critère '
          'devenu néfaste.');
      w('[G4v5]              Cela CONFIRME la propriété demandée par le POINT 4 '
          '(c est bien lui qui est gelé).');
      w('[G4v5]              Cela signifie aussi que ce dispositif NE SÉPARE PAS '
          'les deux notions : il ne permet pas');
      w('[G4v5]              de conclure qu un critère UTILE peut lui aussi être '
          'gelé par la seule règle des poids.');
    } else if (cyclesGelesB > 0) {
      w('[G4v5]            ⇒ COÏNCIDENCE PARTIELLE ($cyclesConcordants/$cyclesGelesB) '
          ': ${cyclesGelesB - cyclesConcordants} cycle(s) gèlent autre chose que '
          'jockey-néfaste.');
      w('[G4v5]              Cela SÉPARE les deux notions et permet de conclure '
          'que le gel suit la règle des poids,');
      w('[G4v5]              et non la perte de pertinence.');
    } else {
      w('[G4v5]            ⇒ AUCUN gel en phase B : le dispositif ne teste pas la '
          'propriété demandée.');
    }
    w('');

    // ═══════════════════════ VERDICT ARGUMENTÉ ═══════════════════════════
    w('[G4v5] ─── VERDICT ARGUMENTÉ ───');
    if (!libreJ || !libreD) {
      w('[G4v5] ⚠️ CONDITIONNEMENT OBLIGATOIRE : G3 a déclaré une SATURATION '
          '(jockey libre=${libreJ ? "OUI" : "NON"} · '
          'distSpec libre=${libreD ? "OUI" : "NON"}).');
      w('[G4v5]   Le verdict ci-dessous ne vaut donc QUE pour la part du trajet '
          'effectuée hors saturation,');
      w('[G4v5]   et NE PROUVE RIEN sur le régime libre demandé par le POINT 4.');
    } else {
      w('[G4v5] ✓ G3 a confirmé un régime LIBRE : le verdict porte bien sur le '
          'cas demandé.');
    }
    final djActif = avecGel.pJ[2 * kPhaseG5 - 1] - avecGel.pJ[kPhaseG5];
    final djSans = sansGel.pJ[2 * kPhaseG5 - 1] - sansGel.pJ[kPhaseG5];
    final ddActif = avecGel.pD[2 * kPhaseG5 - 1] - avecGel.pD[kPhaseG5];
    final ddSans = sansGel.pD[2 * kPhaseG5 - 1] - sansGel.pD[kPhaseG5];
    w('[G4v5] PHASE B — pendant que jockey est NÉFASTE et CORRÉLÉ :');
    w('[G4v5]   jockey   Δ avec gel=${d9(djActif)} · sans gel=${d9(djSans)} · '
        'écart=${d9(djSans - djActif)}');
    w('[G4v5]   distSpec Δ avec gel=${d9(ddActif)} · sans gel=${d9(ddSans)} · '
        'écart=${d9(ddSans - ddActif)}');

    final ecartJ = (djSans - djActif).abs();
    final ecartD = (ddSans - ddActif).abs();
    final seuil = 1e-6;
    if (ecartJ < seuil && ecartD < seuil) {
      w('[G4v5] ⇒ LE GEL NE MODIFIE RIEN, MÊME EN RÉGIME LIBRE. Le gradient de '
          'jockey est coupé,');
      w('[G4v5]   mais la PROJECTION le déplace autant que s il ne l était pas. '
          'Le gel est donc');
      w('[G4v5]   SANS EFFET FONCTIONNEL sur l adaptation, dans ce périmètre.');
    } else {
      w('[G4v5] ⇒ LE GEL MODIFIE L ADAPTATION : écart jockey=${d9(ecartJ)} · '
          'écart distSpec=${d9(ecartD)}.');
      if (djActif.abs() < djSans.abs()) {
        w('[G4v5]   Le critère gelé se déplace MOINS que s il était libre ⇒ '
            'BLOCAGE PARTIEL DÉMONTRÉ.');
        w('[G4v5]   Rapport de blocage |Δ libre| / |Δ gelé| = '
            '${d9(djSans.abs() / (djActif.abs() + 1e-18))}');
      } else {
        w('[G4v5]   Le critère gelé ne se déplace pas moins ⇒ pas de blocage '
            'net dans ce régime.');
      }
    }
    w('');
    w('[G4v5] PHASE C — RÉVERSIBILITÉ (les deux redeviennent UTILES) :');
    w('[G4v5]   jockey   ${d9(avecGel.pJ[2 * kPhaseG5])} → '
        '${d9(avecGel.pJ[3 * kPhaseG5 - 1])} '
        '(Δ=${d9(avecGel.pJ[3 * kPhaseG5 - 1] - avecGel.pJ[2 * kPhaseG5])})');
    w('[G4v5]   distSpec ${d9(avecGel.pD[2 * kPhaseG5])} → '
        '${d9(avecGel.pD[3 * kPhaseG5 - 1])} '
        '(Δ=${d9(avecGel.pD[3 * kPhaseG5 - 1] - avecGel.pD[2 * kPhaseG5])})');
    final remonte = avecGel.pJ[3 * kPhaseG5 - 1] > avecGel.pJ[2 * kPhaseG5] + 1e-9;
    w('[G4v5]   ⇒ jockey REMONTE en phase C : ${remonte ? "OUI" : "NON"}');

    for (final l in out) {
      // ignore: avoid_print
      print(l);
    }
  }, timeout: const Timeout(Duration(minutes: 60)));
}

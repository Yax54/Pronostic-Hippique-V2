const int kPhaseE = 400;
const int kTotalE = 3 * kPhaseE;

/// Profils E1 : le groupe [groupeUtile] porte γ = +20, l'autre γ = −20.
///
/// ⚠️ Chaque critère a sa PROPRE composante latente (β appliqué à un signal
/// dont le sel dépend de l'index), donc aucune composante commune aux 19.
Map<String, Profil> profilsE1({required bool basesUtiles, double gamma = 20.0}) {
  final m = <String, Profil>{};
  for (var i = 0; i < kCles.length; i++) {
    final k = kCles[i];
    final estBase = kBase.contains(k);
    final utile = estBase == basesUtiles;
    m[k] = Profil(beta: 0, gamma: utile ? gamma : -gamma);
  }
  return m;
}

/// Écart-type des 12 poids enrichis : mesure de la REDISTRIBUTION INTERNE.
double ecartTypeEnrichisV4(Map<String, double> m) {
  final v = <double>[for (final k in kEnrichis) m[k] ?? 0.0];
  final moy = v.reduce((a, b) => a + b) / v.length;
  final var2 =
      v.map((x) => (x - moy) * (x - moy)).reduce((a, b) => a + b) / v.length;
  return math.sqrt(var2);
}

double amplitudeEnrichisV4(Map<String, double> m) {
  final v = <double>[for (final k in kEnrichis) m[k] ?? 0.0];
  return v.reduce((a, b) => a > b ? a : b) - v.reduce((a, b) => a < b ? a : b);
}

/// Déroulement d'un essai : trois phases, [gelActif] selon l'essai.
Future<({
  List<double> parts,
  List<double> ets,
  List<double> amps,
  int hors,
  Map<String, double> discDebut,
  Map<String, double> discFin,
  int gelTotal,
})> essaiE1({required bool gelActif, required String nom}) async {
  IaExperimentation.reinitialiser();
  IaExperimentation.gelActif = gelActif;

  final courses = <IaPronostic>[
    for (var i = 0; i < kTotalE; i++)
      pronostic(
        courseKey: 'C${i.toString().padLeft(4, '0')}',
        date: dateCycle4(i, nbCycles: kTotalE),
        resolu: false,
        variance: 6.0 + (i % 5) * 4.0,
        scoresCriteres: <String, ScoresCriteres>{
          for (var n = 1; n <= 5; n++)
            '$n': notesV4(
              n,
              i,
              profilsE1(basesUtiles: i < kPhaseE || i >= 2 * kPhaseE),
            ),
        },
        arrivee: const <int>[1, 2, 3, 4, 5],
      ),
    for (var j = 0; j < 60; j++)
      pronostic(
        courseKey: 'A${j.toString().padLeft(4, '0')}',
        date: dateAmorce4(j, kTotalE),
        variance: 6.0 + (j % 5) * 4.0,
        scoresCriteres: <String, ScoresCriteres>{
          for (var n = 1; n <= 5; n++)
            '$n': notesV4(n, j + 1, profilsE1(basesUtiles: true)),
        },
        arrivee: const <int>[1, 2, 3, 4, 5],
      ),
  ];
  await semer(
    pronostics: courses,
    verifierNombre: false,
    poidsDepart: kInitial,
    sigmaAttendue: sommeCle(kInitial, kCles),
  );

  final parts = <double>[], ets = <double>[], amps = <double>[];
  var hors = 0;
  var gelTotal = 0;
  Map<String, double> discDebut = <String, double>{};
  Map<String, double> discFin = <String, double>{};

  for (var i = 0; i < kTotalE; i++) {
    await executerCycle(
      courseKey: 'C${i.toString().padLeft(4, '0')}',
      arrivee: const <int>[1, 2, 3, 4, 5],
    );
    final p = poidsEffectifs();
    final sigma = sommeCle(p, kCles);
    final part = partEnrichie4(p);
    if ((sigma - 1.0).abs() > 1e-6) hors++;
    if (part < 0.18 - 1e-9 || part > 0.40 + 1e-9) hors++;
    parts.add(part);
    ets.add(ecartTypeEnrichisV4(p));
    amps.add(amplitudeEnrichisV4(p));
    gelTotal += IaExperimentation.derniersGeles.length;
    if (i == 80) discDebut = discriminationsMoteur();
    if (i == kTotalE - 1) discFin = discriminationsMoteur();
  }
  return (
    parts: parts, ets: ets, amps: amps, hors: hors,
    discDebut: discDebut, discFin: discFin, gelTotal: gelTotal,
  );
}

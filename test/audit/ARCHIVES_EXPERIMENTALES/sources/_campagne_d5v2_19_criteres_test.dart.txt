import 'package:flutter_test/flutter_test.dart';

import 'package:pronostic_hippique/services/ia_experimentation.dart';
import 'package:pronostic_hippique/services/ia_memory_models.dart';

import '../support/atelier_apprentissage.dart';
import 'harnais_v3.dart';

const int kCycles = 300;
const int kBascule = 100;
const int kAmorce = 60;

/// Une campagne pour UN critère, dans UN sens.
Future<({
  List<double> poids,
  List<double> disc,
  int geles,
  bool sommeOk,
  bool discVarie,
})> campagneCritere({required String cible, required bool versPlancher}) async {
  IaExperimentation.reinitialiser();
  final poids = <double>[];
  final disc = <double>[];
  var geles = 0;
  var sommeOk = true;

  double pente(int cycle) {
    final phase1 = cycle < kBascule;
    final nuisible = versPlancher ? phase1 : !phase1;
    return nuisible ? -12.0 : 12.0;
  }

  final courses = <IaPronostic>[
    for (var i = 0; i < kCycles; i++)
      pronostic(
        courseKey: 'C${i.toString().padLeft(4, '0')}',
        date: dateCycle3(i, nbCycles: kCycles),
        resolu: false,
        variance: 6.0 + (i % 5) * 4.0,
        scoresCriteres: <String, ScoresCriteres>{
          for (var n = 1; n <= 5; n++)
            '$n': notesPilotees(n, i, pilote: cible, pente: pente(i)),
        },
        arrivee: const <int>[1, 2, 3, 4, 5],
      ),
    // Amorçage : 60 courses DÉJÀ résolues, notes variées (cycle j+1) pour
    // qu'elles ne soient pas toutes identiques. Elles occupent la fenêtre au
    // début puis en sortent naturellement, la date étant plus ancienne.
    for (var j = 0; j < kAmorce; j++)
      pronostic(
        courseKey: 'A${j.toString().padLeft(4, '0')}',
        date: dateAmorce3(j, kCycles),
        variance: 6.0 + (j % 5) * 4.0,
        scoresCriteres: <String, ScoresCriteres>{
          for (var n = 1; n <= 5; n++)
            '$n': notesPilotees(n, 200 + j, pilote: cible, pente: pente(j)),
        },
        arrivee: const <int>[1, 2, 3, 4, 5],
      ),
  ];
  await semer(
    pronostics: courses,
    verifierNombre: false,
    poidsDepart: kInitial,
    sigmaAttendue: sommeCle(kInitial, kCles),
  );

  for (var i = 0; i < kCycles; i++) {
    await executerCycle(
      courseKey: 'C${i.toString().padLeft(4, '0')}',
      arrivee: const <int>[1, 2, 3, 4, 5],
    );
    final p = poidsEffectifs();
    if ((sommeCle(p, kCles) - 1.0).abs() > 1e-6) sommeOk = false;
    poids.add(p[cible]!);
    disc.add(IaExperimentation.dernieresDiscriminations[cible] ?? 0.0);
    if (IaExperimentation.derniersGeles.contains(cible)) geles++;
  }

  final discVarie = disc.sublist(kBascule).toSet().length > 1;
  return (
    poids: poids,
    disc: disc,
    geles: geles,
    sommeOk: sommeOk,
    discVarie: discVarie,
  );
}

void main() {
  TestWidgetsFlutterBinding.ensureInitialized();

  test('D5v2 — 19 critères × (plancher, plafond) · $kCycles cycles', () async {
    final bornes = IaPoidsAdaptatifs.bornesIndividuelles();
    final lignes = <String>[];
    var pAtteint = 0, pQuitte = 0, pValide = 0;
    var hAtteint = 0, hQuitte = 0, hValide = 0;
    var invalides = <String>[];

    // ignore: avoid_print
    print('');
    // ignore: avoid_print
    print('[D5v2] ═══ 19 CRITÈRES · $kCycles CYCLES · BASCULE $kBascule · '
        'AMORCE $kAmorce ═══');
    // ignore: avoid_print
    print('[D5v2] harnais v3 (V0ter : |r|max 0,088187 · témoins |d|<0,025 · '
        'piloté ±0,360000000)');

    var idx = 0;
    for (final k in kCles) {
      idx++;
      final lo = bornes[k]![0];
      final hi = bornes[k]![1];

      final p = await campagneCritere(cible: k, versPlancher: true);
      final mini = p.poids.sublist(0, kBascule).reduce((a, b) => a < b ? a : b);
      final finP = p.poids.last;
      final atPLo = (mini - lo).abs() < 1e-9;
      final quitteLo = finP > lo + 1e-9;
      if (atPLo) pAtteint++;
      if (quitteLo) pQuitte++;
      final valideP = p.sommeOk && p.discVarie;
      if (valideP) pValide++;
      if (!valideP) invalides.add('$k/plancher');

      final h = await campagneCritere(cible: k, versPlancher: false);
      final maxi = p.poids.isEmpty
          ? 0.0
          : h.poids.sublist(0, kBascule).reduce((a, b) => a > b ? a : b);
      final finH = h.poids.last;
      final atHi = (maxi - hi).abs() < 1e-6;
      final quitteHi = finH < hi - 1e-9;
      if (atHi) hAtteint++;
      if (quitteHi) hQuitte++;
      final valideH = h.sommeOk && h.discVarie;
      if (valideH) hValide++;
      if (!valideH) invalides.add('$k/plafond');

      lignes.add('[D5v2] ${k.padRight(12)} lo=${lo.toStringAsFixed(3)} '
          'mini=${d9(mini)} fin=${d9(finP)} '
          '${atPLo ? "ATTEINT" : "-------"} ${quitteLo ? "QUITTE" : "------"} '
          'g=${p.geles.toString().padLeft(3)} | '
          'hi=${hi.toStringAsFixed(2)} maxi=${d9(maxi)} fin=${d9(finH)} '
          '${atHi ? "ATTEINT" : "-------"} ${quitteHi ? "QUITTE" : "------"} '
          'g=${h.geles.toString().padLeft(3)}');

      // Progression visible pour pouvoir estimer le temps restant.
      if (idx % 5 == 0) {
        // ignore: avoid_print
        print('[D5v2] … $idx/19 critères traités');
      }
    }

    for (final l in lignes) {
      // ignore: avoid_print
      print(l);
    }
    // ignore: avoid_print
    print('[D5v2] ⇒ PLANCHER : atteint $pAtteint/19 · quitté $pQuitte/19 · '
        'campagnes valides $pValide/19');
    // ignore: avoid_print
    print('[D5v2] ⇒ PLAFOND  : atteint $hAtteint/19 · quitté $hQuitte/19 · '
        'campagnes valides $hValide/19');
    if (invalides.isNotEmpty) {
      // ignore: avoid_print
      print('[D5v2] ⇒ CAMPAGNES INVALIDES (signal non atteint) : '
          '${invalides.join(", ")}');
    }

    // On n'exige PAS un résultat : on exige que toute campagne rapportée soit
    // valide. Un résultat « borne non atteinte » est une mesure recevable.
    expect(pValide + hValide, 38,
        reason: 'Toute campagne dont Σ ≠ 1 ou dont la discrimination ne varie '
            'pas doit être exclue de l interprétation. Invalidées : '
            '${invalides.join(", ")}');
  }, timeout: const Timeout(Duration(minutes: 300)));
}

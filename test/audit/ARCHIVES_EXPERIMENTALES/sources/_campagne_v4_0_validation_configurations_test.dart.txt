import 'package:flutter_test/flutter_test.dart';

import '../support/atelier_apprentissage.dart';
import 'harnais_v4.dart';
import 'validation_prealable.dart';

/// Profils neutres, éventuellement écrasés.
Map<String, Profil> profilsV4(Map<String, Profil> remplacements) {
  final m = profilsNeutres();
  m.addAll(remplacements);
  return m;
}

void main() {
  TestWidgetsFlutterBinding.ensureInitialized();

  test('V4-0/1 — POSITIVE forte (C1 : β=25 γ=20 des deux côtés)', () async {
    final spec = Spec(
      nom: 'C1 paire POSITIVE forte (β=25 γ=20)',
      profils: profilsV4(<String, Profil>{
        'distSpec': const Profil(beta: 25, gamma: 20),
        'jockey': const Profil(beta: 25, gamma: 20),
      }),
      attentesPaires: const <AttentePaire>[
        AttentePaire('distSpec', 'jockey', minAbs: 0.85),
      ],
      attentesDiscriminations: const <AttenteDiscrimination>[
        AttenteDiscrimination('distSpec', signe: 1, minAbs: 0.034),
      ],
      gradientAttendu: const <String>['distSpec', 'jockey'],
    );
    final v = await valider(spec, cycles: 45);
    v.imprimer('V4-0/1');
    expect(v.ok, isTrue, reason: 'C1 doit être acceptée');
  }, timeout: const Timeout(Duration(minutes: 90)));

  test('V4-0/2 — NÉGATIVE forte (C2 : utile γ=+20 vs néfaste γ=−20)',
      () async {
    // ⚠️ Cette configuration est ANTI-corrélée PARCE QUE les deux critères
    // sont de pertinence OPPOSÉE. Le moteur, qui stocke |r|, la voit comme une
    // redondance à 0,99 : c'est précisément le comportement à documenter.
    final spec = Spec(
      nom: 'C2 paire ANTI-corrélée (utile γ=+20 / néfaste γ=−20)',
      profils: profilsV4(<String, Profil>{
        'distSpec': const Profil(beta: 0, gamma: 20),
        'jockey': const Profil(beta: 0, gamma: -20),
      }),
      attentesPaires: const <AttentePaire>[
        AttentePaire('distSpec', 'jockey', minAbs: 0.85),
      ],
      attentesDiscriminations: const <AttenteDiscrimination>[
        AttenteDiscrimination('distSpec', signe: 1, minAbs: 0.034),
        AttenteDiscrimination('jockey', signe: -1, minAbs: 0.042),
      ],
      gradientAttendu: const <String>['distSpec', 'jockey'],
    );
    final v = await valider(spec, cycles: 45);
    v.imprimer('V4-0/2');
    expect(v.ok, isTrue, reason: 'C2 doit être acceptée');
  }, timeout: const Timeout(Duration(minutes: 90)));

  test('V4-0/3 — NON corrélée (C3 : utile vs neutre)', () async {
    final spec = Spec(
      nom: 'C3 paire NON corrélée (utile γ=20 · neutre)',
      profils: profilsV4(<String, Profil>{
        'distSpec': const Profil(beta: 0, gamma: 20),
        'jockey': const Profil(beta: 0, gamma: 0),
      }),
      attentesPaires: const <AttentePaire>[
        AttentePaire('distSpec', 'jockey', minAbs: 0.0, maxAbs: 0.60),
      ],
      attentesDiscriminations: const <AttenteDiscrimination>[
        AttenteDiscrimination('distSpec', signe: 1, minAbs: 0.034),
      ],
      gradientAttendu: const <String>['distSpec', 'jockey'],
    );
    final v = await valider(spec, cycles: 45);
    v.imprimer('V4-0/3');
    expect(v.ok, isTrue, reason: 'C3 doit être acceptée');
  }, timeout: const Timeout(Duration(minutes: 90)));

  test('V4-0/4 — DISCRIMINATIONS DIFFÉRENTES (C4 : γ=25 / γ=9)', () async {
    final spec = Spec(
      nom: 'C4 paire corrélée, discriminations différentes (γ=25 / γ=9)',
      profils: profilsV4(<String, Profil>{
        'distSpec': const Profil(beta: 35, gamma: 25),
        'jockey': const Profil(beta: 35, gamma: 9),
      }),
      attentesPaires: const <AttentePaire>[
        AttentePaire('distSpec', 'jockey', minAbs: 0.78),
      ],
      attentesDiscriminations: const <AttenteDiscrimination>[
        AttenteDiscrimination('distSpec', signe: 1, minAbs: 0.042),
        AttenteDiscrimination('jockey', signe: 1, minAbs: 0.013),
      ],
      gradientAttendu: const <String>['distSpec', 'jockey'],
    );
    final v = await valider(spec, cycles: 45);
    v.imprimer('V4-0/4');
    expect(v.ok, isTrue, reason: 'C4 doit être acceptée');
  }, timeout: const Timeout(Duration(minutes: 90)));

  test('V4-0/5 — critère effectivement GELÉ (C1)', () async {
    final spec = Spec(
      nom: '5. critère effectivement GELÉ (C1)',
      profils: profilsV4(<String, Profil>{
        'distSpec': const Profil(beta: 25, gamma: 20),
        'jockey': const Profil(beta: 25, gamma: 20),
      }),
      attentesPaires: const <AttentePaire>[
        AttentePaire('distSpec', 'jockey', minAbs: 0.85),
      ],
      gelAttendu: 'jockey',
      gradientAttendu: const <String>['distSpec'],
    );
    final v = await valider(spec, cycles: 45);
    v.imprimer('V4-0/5');
    // Le gel doit être ATTEINT : sans cela G4v3 n'a rien à mesurer.
    // Le critère désigné par la règle `p1 <= p2 ? c1 : c2` doit être jockey
    // (0,07 < 0,08 au départ), ce que `valider` vérifie explicitement.
    expect(v.ok, isTrue, reason: 'le gel doit être atteint sur jockey');
  }, timeout: const Timeout(Duration(minutes: 90)));

  test('V4-0/7 — INVERSION des poids initiaux (C1 sur deux départs)', () async {
    final profils = profilsV4(<String, Profil>{
      'distSpec': const Profil(beta: 25, gamma: 20),
      'jockey': const Profil(beta: 25, gamma: 20),
    });

    final specA = Spec(
      nom: '7a. inversion — A (distSpec 0,08 > jockey 0,07)',
      profils: profils,
      attentesPaires: const <AttentePaire>[
        AttentePaire('distSpec', 'jockey', minAbs: 0.85),
      ],
      gelAttendu: 'jockey',
      gradientAttendu: const <String>['distSpec'],
    );
    final va = await valider(specA, cycles: 45);
    va.imprimer('V4-0/7a');

    final departB = <String, double>{...kInitial}
      ..['distSpec'] = 0.03
      ..['jockey'] = 0.09;
    final specB = Spec(
      nom: '7b. inversion — B (distSpec 0,03 < jockey 0,09)',
      profils: profils,
      depart: departB,
      attentesPaires: const <AttentePaire>[
        AttentePaire('distSpec', 'jockey', minAbs: 0.85),
      ],
      gelAttendu: 'distSpec',
      gradientAttendu: const <String>['jockey'],
    );
    final vb = await valider(specB, cycles: 45);
    vb.imprimer('V4-0/7b');

    // ignore: avoid_print
    print('[V4-0/7] ⇒ A gèle jockey · B gèle distSpec ⇒ '
        '${va.ok && vb.ok ? "L INVERSION EST MESURÉE" : "REFUS"}');
    expect(va.ok && vb.ok, isTrue);
  }, timeout: const Timeout(Duration(minutes: 180)));
}

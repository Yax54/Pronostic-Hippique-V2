const int kPhaseG4 = 200;
const int kTotalG4 = 4 * kPhaseG4;

/// Profils de la phase [cycle].
///
/// ⚠️ ERREUR DE CONCEPTION MESURÉE PUIS CORRIGÉE — la phase 2 visait à faire
/// TOMBER la corrélation en mettant jockey à β = 0 et γ = 0. La garde a
/// REFUSÉ la configuration : |r| mesuré = 0,9884, identique à la phase 1.
/// Cause : distSpec conservait β = 25. Or un critère neutre (γ = 0) qui porte
/// un signal latent partagé est FORTEMENT corrélé à un autre critère qui porte
/// le même signal : la corrélation ne vient pas de γ mais de β partagé. Mettre
/// jockey à zéro ne le décorrèle donc pas du tout.
/// CORRECTION : pour décorréler, il faut que jockey n ait PLUS la composante
/// commune (β_jockey = 0) ET que son bruit propre soit LARGE (σ = 45), de sorte
/// que sa variance soit dominée par du bruit indépendant. La corrélation entre
/// un signal commun d'amplitude 25 et un bruit indépendant d'amplitude 45 vaut
/// alors 25/√(25²+45²) ≈ 0,49, en théorie. Le moteur tranchera ; la garde
/// exige |r| ≤ 0,60 et REFUSERA si ce n'est pas atteint.
({double betaD, double gammaD, double sigmaD, double betaJ, double gammaJ,
    double sigmaJ, String nom}) profilsPhase(int cycle) {
  if (cycle < kPhaseG4) {
    return (betaD: 25, gammaD: -20, sigmaD: 6, betaJ: 25, gammaJ: -20, sigmaJ: 6,
        nom: 'P1 NÉFASTE+GELÉ');
  }
  if (cycle < 2 * kPhaseG4) {
    return (betaD: 25, gammaD: -20, sigmaD: 6, betaJ: 0, gammaJ: 0, sigmaJ: 45,
        nom: 'P2 DÉGEL (r≈0)');
  }
  if (cycle < 3 * kPhaseG4) {
    return (betaD: 25, gammaD: -20, sigmaD: 6, betaJ: 25, gammaJ: -20, sigmaJ: 6,
        nom: 'P3 REPRISE DU GEL');
  }
  return (betaD: 25, gammaD: 20, sigmaD: 6, betaJ: 25, gammaJ: 20, sigmaJ: 6,
      nom: 'P4 UTILES (réversibilité)');
}

Map<String, Profil> profilsDe(int cycle) {
  final p = profilsPhase(cycle);
  return profilsV4G4(<String, Profil>{
    'distSpec': Profil(beta: p.betaD, gamma: p.gammaD, sigma: p.sigmaD),
    'jockey': Profil(beta: p.betaJ, gamma: p.gammaJ, sigma: p.sigmaJ),
  });
}

Map<String, Profil> profilsV4G4(Map<String, Profil> r) {
  final m = profilsNeutres();
  m.addAll(r);
  return m;
}

      nom: 'G4v3 P2 décorrélé (β_jockey=0, γ_jockey=0, σ_jockey=45)',
      profils: profilsDe(kPhaseG4),
      attentesPaires: const <AttentePaire>[
        AttentePaire('distSpec', 'jockey', minAbs: 0.0, maxAbs: 0.60),
      ],
      attentesDiscriminations: const <AttenteDiscrimination>[
        AttenteDiscrimination('distSpec', signe: -1, minAbs: 0.034),
      ],
      gradientAttendu: const <String>['distSpec', 'jockey'],
    );
    final vP2 = await valider(specP2, cycles: 45);
    vP2.imprimer('G4v3/P2');

    // ─── La campagne longue ne démarre QUE si les deux gardes passent ───────
    expect(vP1.ok, isTrue,
        reason: 'P1 doit corréler ≥ 0,85 ET geler jockey, sinon la mesure du '

    expect(vP2.ok, isTrue,
        reason: 'P2 doit réellement décorréler (|r| ≤ 0,60), sinon le dégel '

    IaExperimentation.reinitialiser();

          },

        pronostic(
          courseKey: 'A${j.toString().padLeft(4, '0')}',

          },

    await semer(

      poidsDepart: kInitial,

    );

    // ─── Trace par cycle ────────────────────────────────────────────────────
    final pJ = <double>[], pD = <double>[];
    final gJ = <double>[], gD = <double>[];
    final rPaire = <double>[];
    final gelJ = <int>[], gelD = <int>[];
    final selJ = <int>[];

    for (var i = 0; i < kTotalG4; i++) {
      await executerCycle(

      selJ.add(p['distSpec']! <= p['jockey']! ? 1 : 0);
    }

    int somme(List<int> v, int a, int b) =>
        v.sublist(a, b).fold(0, (x, y) => x + y);
    double moy(List<double> v, int a, int b) {
      final s = v.sublist(a, b);
      return s.reduce((x, y) => x + y) / s.length;
    }

    double maxAbs(List<double> v, int a, int b) {
      var m = 0.0;
      for (final x in v.sublist(a, b)) {
        if (x.abs() > m) m = x.abs();
      }
      return m;
    }

    print('');

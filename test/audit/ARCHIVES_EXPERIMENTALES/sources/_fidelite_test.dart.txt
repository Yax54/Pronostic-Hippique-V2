import 'package:flutter_test/flutter_test.dart';

import 'package:pronostic_hippique/services/ia_experimentation.dart';

ScoresCriteres notesAlignees(int numero) => ScoresCriteres(
      forme: 96.0 - numero * 6, gains: 96.0 - numero * 6,
      record: 96.0 - numero * 6, cote: 96.0 - numero * 6,
      constance: 96.0 - numero * 6, victoires: 96.0 - numero * 6,
      discipline: 96.0 - numero * 6, distSpec: 96.0 - numero * 6,
      jockey: 96.0 - numero * 6, repos: 96.0 - numero * 6,
      hippo: 96.0 - numero * 6, entraineur: 96.0 - numero * 6,
      elo: 96.0 - numero * 6, terrain: 96.0 - numero * 6,
      divergence: 96.0 - numero * 6, poidsRel: 96.0 - numero * 6,
      progression: 96.0 - numero * 6, mouvCote: 96.0 - numero * 6,
      placeDepart: 96.0 - numero * 6,
    );

void main() {
  TestWidgetsFlutterBinding.ensureInitialized();

  test('FID — paramètres de production ⇒ trajectoire identique', () async {
    IaExperimentation.reinitialiser();

    final courses = <IaPronostic>[
      for (var i = 0; i < 20; i++)
        pronostic(
          courseKey: 'C${i.toString().padLeft(4, '0')}',
          date: DateTime.now().subtract(Duration(days: 2 + i)),
          resolu: false,
          scoresCriteres: <String, ScoresCriteres>{
            for (var n = 1; n <= 5; n++) '$n': notesAlignees(n),
          },
        ),
      for (var j = 0; j < 12; j++)
        pronostic(
          courseKey: 'A${j.toString().padLeft(4, '0')}',
          date: DateTime.now().subtract(Duration(days: 22 + j)),
          scoresCriteres: <String, ScoresCriteres>{
            for (var n = 1; n <= 5; n++) '$n': notesAlignees(n),
          },
        ),
    ];
    await semer(pronostics: courses, verifierNombre: false);

    // 20 cycles réels, suite déterministe.
    for (var i = 0; i < 20; i++) {
      await executerCycle(
        courseKey: 'C${i.toString().padLeft(4, '0')}',
        arrivee: const <int>[1, 2, 3, 4, 5],
      );
    }

    final p = poidsEffectifs();
    final ligne = <String>[
      for (final k in kCles) '${p[k]!.toStringAsFixed(15)}',
    ].join(' ');

    // ignore: avoid_print
    print('[FID] EMPREINTE=$ligne');
    // ignore: avoid_print
    print('[FID] nbMisesAJour=${IaMemoryService.instance.poids.nbMisesAJour}');
    // ignore: avoid_print
    print('[FID] compteurGradient=${IaExperimentation.compteurGradient}');
    // ignore: avoid_print
    print('[FID] lr=${IaExperimentation.dernierLr}');

    expect(IaMemoryService.instance.poids.nbMisesAJour, 20);
  });
}

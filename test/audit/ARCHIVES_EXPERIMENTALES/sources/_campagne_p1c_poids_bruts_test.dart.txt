// ───────────────────────────────────────────────────────────────────────────
//  SCORE BRUT — recomposé avec les grandeurs PUBLIÉES par la production
// ───────────────────────────────────────────────────────────────────────────
//
//  `analyserCourseAvecCriteres` publie les `ScoresCriteres` par cheval. Les
//  poids effectifs sont exposés par `poidsEffectifsPourDiscipline()`, exactement
//  le `pe` que `_calculerScoreBrutEtCriteres` emploie. La composition est la
//  MÊME expression que la production : Σ note×poids, SANS division par Σ
//  (cf. ia_pronostic_engine.dart:391-409). Ce n'est donc pas un second moteur.
double scoreBrutRecompose(ScoresCriteres sc, Map<String, double> pe) {
  var s = 0.0;
  for (final k in kCles) {
    s += sc.valeurPourCritere(k) * (pe[k] ?? 0.0);
  }
  return s;
}

String d9(double v) => v.toStringAsFixed(9);

/// Ordonne les numéros par score décroissant (l'ordre du moteur).
List<int> ordrePar(Map<int, double> m) {
  final ks = m.keys.toList();
  ks.sort((a, b) => m[b]!.compareTo(m[a]!));
  return ks;
}

int pairesInversees(List<int> a, List<int> b) {
  expect(a.length, b.length);
  var n = 0;
  for (var i = 0; i < a.length; i++) {
    for (var j = i + 1; j < a.length; j++) {
      final ia = a.indexOf(a[i]);
      final ja = a.indexOf(a[j]);
      final ib = b.indexOf(a[i]);
      final jb = b.indexOf(a[j]);
      if ((ia < ja) != (ib < jb)) n++;
    }
  }
  return n;
}

    w('[P1C] ordre par score PUBLIÉ   BRUT    : ${ordrePublie1.join(" > ")}');
    w('[P1C] ordre par score PUBLIÉ   PROJETÉ : ${ordrePublie2.join(" > ")}');
    w('[P1C] ordre par score BRUT     BRUT    : ${ordreBrut1.join(" > ")}');
    w('[P1C] ordre par score BRUT     PROJETÉ : ${ordreBrut2.join(" > ")}');
    w('[P1C] paires inversées (publié) : ${pairesInversees(ordrePublie1, ordrePublie2)}/28');
    w('[P1C] paires inversées (brut)   : ${pairesInversees(ordreBrut1, ordreBrut2)}/28');
    final memeOrdrePublie = ordrePublie1.join(',') == ordrePublie2.join(',');
    w('[P1C] ordre publié identique ? ${memeOrdrePublie ? "OUI" : "NON"}');
    w('');

    // ─── 5. RECOMMANDATIONS
    w('[P1C] ─── 5. TOP5 · SÉLECTION · CONFIANCE · CONSEIL ───');
    w('[P1C] top5 BRUT    : ${etat1.top5.join(" - ")}');
    w('[P1C] top5 PROJETÉ : ${etat2.top5.join(" - ")}');
    w('[P1C] sélection BRUT    : ${etat1.selection}');
    w('[P1C] sélection PROJETÉ : ${etat2.selection}');
    w('[P1C] confiance BRUT    : ${d9(etat1.confiance)}');
    w('[P1C] confiance PROJETÉ : ${d9(etat2.confiance)}');
    w('[P1C] conseil BRUT    : ${etat1.conseil}');
    w('[P1C] conseil PROJETÉ : ${etat2.conseil}');
    w('[P1C] conseil identique ? '
        '${etat1.conseil == etat2.conseil ? "OUI" : "NON"}');
    w('');

    // ─── 6. PREMIÈRE TRAJECTOIRE D'APPRENTISSAGE
    //  Un cycle RÉEL depuis CHACUN des deux états. On mesure les poids obtenus,
    //  la variation par critère, et l'état du signal (momentum) reçu.
    w('[P1C] ─── 6. PREMIÈRE TRAJECTOIRE D APPRENTISSAGE ───');
    final histo = <IaPronostic>[
      for (var i = 0; i < 12; i++)
        pronostic(
          courseKey: 'P1C_H$i',
          date: DateTime(2026, 11, 20).subtract(Duration(days: 12 - i)),
        ),
    ];

    Future<List<String>> trajectoire(Map<String, double> depart) async {
      await semer(
        pronostics: <IaPronostic>[
          ...histo,
          pronostic(
            courseKey: 'P1C_CIBLE',
            date: DateTime(2026, 11, 21),
            resolu: false,
          ),
        ],
        poidsDepart: depart,
        sigmaAttendue: sommeCle(depart, kCles),
      );
      final av = <String, double>{
        for (final k in kCles) k: IaMemoryService.instance.poids.getPoids(k),
      };
      final c = await executerCycle(
        courseKey: 'P1C_CIBLE',
        arrivee: <int>[1, 2, 3, 4, 5],
      );
      final res = <String>[];
      res.add('  aAppris=${c.aAppris} · nbMisesAJour=${c.nbMisesAJour}');
      res.add('  critère      avant        proposé      obtenu      '
          'Δproposé     Δobtenu     momentum');
      for (final k in kCles) {
        res.add('  ${k.padRight(12)} ${d9(av[k]!).padLeft(11)} '
            '${d9(c.propose[k]!).padLeft(12)} ${d9(c.obtenu[k]!).padLeft(12)} '
            '${d9(c.propose_(k)).padLeft(12)} ${d9(c.obtenu_(k)).padLeft(12)} '
            '${d9(c.momentum[k] ?? 0.0).padLeft(12)}');
      }
      var sAv = 0.0, sOb = 0.0, sMo = 0.0;
      var nz = 0;
      for (final k in kCles) {
        sAv += av[k]!;
        sOb += c.obtenu[k]!;
        sMo += (c.momentum[k] ?? 0.0).abs();
        if ((c.momentum[k] ?? 0.0).abs() > 1e-12) nz++;
      }
      res.add('  Σ avant=${d9(sAv)} Σ obtenu=${d9(sOb)} · critères avec '
          'momentum non nul : $nz/19 · Σ|momentum|=${d9(sMo)}');
      return res;
    }

    w('[P1C] -- trajectoire depuis BRUT --');
    for (final l in await trajectoire(kInitial)) {
      w('[P1C] $l');
    }
    w('[P1C] -- trajectoire depuis PROJETÉ --');
    for (final l in await trajectoire(projete)) {
      w('[P1C] $l');
    }

    for (final l in out) {
      // ignore: avoid_print
      print(l);
    }
  }, timeout: const Timeout(Duration(minutes: 15)));
}
